    56→const EMO_PER_PAGE = 40;
    57→
    58→export default function ChatWindow(p: Props) {
    59→  const [messages, setMessages] = useState<ChatMessage[]>(p.seed);
    60→  const [canSend, setCanSend] = useState(false);
    61→  const [pickerOpen, setPickerOpen] = useState(false);
    62→  const [sidebar, setSidebar] = useState(true);
    63→  const [formatOpen, setFormatOpen] = useState(false);
    64→  const [audiblesOpen, setAudiblesOpen] = useState(false);
    65→  const [audibleCat, setAudibleCat] = useState(AUDIBLE_CATEGORIES[0].id);
    66→  const [catMenuOpen, setCatMenuOpen] = useState(false);
    67→  const [selAudible, setSelAudible] = useState<string | null>(null);
    68→  const [emoPage, setEmoPage] = useState(0);
    69→  const [imv, setImv] = useState<ImvId>("none");
    70→  const [imvMenu, setImvMenu] = useState(false);
    71→  const [gust, setGust] = useState(0);
    72→  const [kisses, setKisses] = useState<number[]>([]);
    73→  const [burst, setBurst] = useState(0);
    74→  const [flurry, setFlurry] = useState(0);
    75→  const [shake, setShake] = useState(false);
    76→  const [font, setFont] = useState("Arial");
    77→  const [fsize, setFsize] = useState(10);
    78→  const [bold, setBold] = useState(false);
    79→  const [italic, setItalic] = useState(false);
    80→  const [underline, setUnderline] = useState(false);
    81→  const editor = useRef<HTMLDivElement>(null);
    82→  const pane = useRef<HTMLDivElement>(null);
    83→  const replyTimer = useRef<number | null>(null);
    84→
    85→  useEffect(() => {
    86→    pane.current?.scrollTo({ top: pane.current.scrollHeight });
    87→  }, [messages]);
    88→
    89→  useEffect(
    90→    () => () => {
    91→      if (replyTimer.current) window.clearTimeout(replyTimer.current);
    92→    },
    93→    [],
    94→  );
    95→
    96→  const push = (m: ChatMessage) => setMessages((prev) => [...prev, m]);
    97→
    98→  const doBuzz = () => {
    99→    if (imv === "luv") {
   100→      const id = Date.now();
   101→      setKisses((k) => [...k, id]);
   102→      window.setTimeout(() => setKisses((k) => k.filter((x) => x !== id)), 3400);
   103→      playKiss();
   104→      setMessages((prev) => [...prev, { from: "me", buzz: true }]);
   105→      return;
   106→    }
   107→    if (imv === "autumn") {
   108→      setGust((g) => g + 1);
   109→      playWind();
   110→      setMessages((prev) => [...prev, { from: "me", buzz: true }]);
   111→      return;
   112→    }
   113→    if (imv === "fireworks") {
   114→      /* the buzz becomes a grand golden salute over the skyline */
   115→      setBurst((b) => b + 1);
   116→      playBuzz();
   117→      setMessages((prev) => [...prev, { from: "me", buzz: true }]);
   118→      return;
   119→    }
   120→    if (imv === "winter") {
   121→      setFlurry((f) => f + 1);
   122→      playWind();
   123→      setMessages((prev) => [...prev, { from: "me", buzz: true }]);
   124→      return;
   125→    }
   126→    setMessages((prev) => [...prev, { from: "me", buzz: true }]);
   127→    setShake(true);
   128→    playBuzz();
   129→    window.setTimeout(() => setShake(false), 620);
   130→  };
   131→
   132→  const firstBuzz = useRef(true);
   133→  useEffect(() => {
   134→    if (firstBuzz.current) {
   135→      firstBuzz.current = false;
   136→      return;
   137→    }
   138→    doBuzz();
   139→    // eslint-disable-next-line react-hooks/exhaustive-deps
   140→  }, [p.buzzSignal]);
   141→
   142→  /* keep B/I/U button state in sync with the caret selection */
   143→  useEffect(() => {
   144→    const sync = () => {
   145→      const el = editor.current;
   146→      const sel = document.getSelection();
   147→      if (!el || !sel || !sel.anchorNode || !el.contains(sel.anchorNode)) return;
   148→      try {
   149→        setBold(document.queryCommandState("bold"));
   150→        setItalic(document.queryCommandState("italic"));
   151→        setUnderline(document.queryCommandState("underline"));
   152→      } catch {
   153→        /* noop */
   154→      }
   155→    };
   156→    document.addEventListener("selectionchange", sync);
   157→    return () => document.removeEventListener("selectionchange", sync);
   158→  }, []);
   159→
   160→  const applyFont = (f: string) => {
   161→    setFont(f);
   162→    const el = editor.current;
   163→    if (!el) return;
   164→    el.style.fontFamily = f;
   165→    el.focus();
   166→    document.execCommand("fontName", false, f);
   167→  };
   168→
   169→  const applySize = (s: number) => {
   170→    setFsize(s);
   171→    const el = editor.current;
   172→    if (!el) return;
   173→    el.style.fontSize = `${SIZE_PX[s] ?? 13}px`;
   174→    el.focus();
   175→    const legacy = SIZES.indexOf(s);
   176→    if (legacy > 0) document.execCommand("fontSize", false, String(legacy));
   177→  };
   178→
   179→  const toggle = (cmd: string) => {
   180→    editor.current?.focus();
   181→    document.execCommand(cmd, false);
   182→    try {
   183→      setBold(document.queryCommandState("bold"));
   184→      setItalic(document.queryCommandState("italic"));
   185→      setUnderline(document.queryCommandState("underline"));
   186→    } catch {
   187→      /* noop */
   188→    }
   189→  };
   190→
   191→  const send = () => {
   192→    const el = editor.current;
   193→    if (!el || !el.textContent?.trim()) return;
   194→    const inner = emojifyHtml(el.innerHTML);
   195→    /* default settings render exactly like incoming messages (Arial @13px) */
   196→    const custom = font !== DEFAULT_FONT || fsize !== DEFAULT_SIZE;
   197→    push({
   198→      from: "me",
   199→      html: custom ? `<span style="font-family:${font};font-size:${SIZE_PX[fsize] ?? 13}px">${inner}</span>` : inner,
   200→    });
   201→    playSent();
   202→    el.innerHTML = "";
   203→    el.style.fontFamily = font;
   204→    el.style.fontSize = `${SIZE_PX[fsize] ?? 13}px`;
   205→    setCanSend(false);
   206→    el.focus();

   208→    replyTimer.current = window.setTimeout(() => {
   209→      const reply = AUTO_REPLIES[Math.floor(Math.random() * AUTO_REPLIES.length)];
   210→      setMessages((prev) => [...prev, { from: "them", text: reply }]);
   211→      playMessage();
   212→      p.onNotify?.();
   213→    }, 1100 + Math.random() * 1600);
   214→  };
   215→
   216→  const insertEmoticon = (code: string) => {
   217→    editor.current?.focus();
   218→    document.execCommand("insertText", false, code + " ");
   219→    setPickerOpen(false);
   220→    setCanSend(true);
   221→  };
   222→
   223→  const menus: MenuDef[] = [
   224→    {
   225→      label: "Conversation",
   226→      items: [
   227→        { label: "Send", onClick: send },
   228→        { label: "Send a BUZZ!!!", onClick: doBuzz },
   229→        { sep: true },
   230→        { label: "Invite to Conference...", onClick: () => {} },
   231→        { sep: true },
   232→        { label: "Close", onClick: p.onClose },
   233→      ],
   234→    },
   235→    {
   236→      label: "Edit",
   237→      items: [
   238→        { label: "Bold", onClick: () => toggle("bold") },
   239→        { label: "Italic", onClick: () => toggle("italic") },
   240→        { label: "Underline", onClick: () => toggle("underline") },
   241→        { sep: true },
   242→        { label: "Select All", onClick: () => editor.current?.focus() },
   243→      ],
   244→    },
   245→    {
   246→      label: "View",
   247→      items: [
   248→        { label: "Show Contact Sidebar", checked: sidebar, onClick: () => setSidebar((v) => !v) },
   249→        { label: "Text Formatting Bar", checked: formatOpen, onClick: () => setFormatOpen((v) => !v) },
   250→        { label: "News Ticker", checked: true, onClick: () => {} },
   251→      ],
   252→    },
   253→    {
   254→      label: "Actions",
   255→      items: [
   256→        { label: "Video Call", onClick: () => {} },
   257→        { label: "Voice Call", onClick: () => {} },
   258→        { sep: true },
   259→        { label: "Send File...", onClick: () => {} },
   260→        { label: "Share Photos...", onClick: () => {} },
   261→      ],
   262→    },
   263→    {
   264→      label: "Help",
   265→      items: [{ label: "About Yahoo! Messenger", onClick: () => {} }],
   266→    },
   267→  ];
   268→
   269→  const tickerText = NEWS_TICKER.join("   •   ");
   270→
   271→  return (
   272→    <WindowFrame
   273→      title={p.contact.name}
   274→      icon={<StatusDot status={p.contact.status} />}
   275→      x={p.x}

   277→      width={p.width}
   278→      height={p.height}
   279→      minW={450}
   280→      minH={420}
   281→      onResize={p.onResize}
   282→      z={p.z}
   283→      focused={p.focused}
   284→      maximized={p.maximized}
   285→      shaking={shake}
   286→      menus={menus}

   292→    >
   293→      {/* ---- Call toolbar ---- */}
   294→      <div className="bg-[#f4f3ea] border-b border-[#d8d5c5] px-2 py-1.5 flex items-start">
   295→        <ToolBtn icon={<IconVideoCam />} label="Video Call" />
   296→        <ToolBtn icon={<IconMic />} label="Voice Call" caret />
   297→        <div className="flex-1" />
   298→        <div className="relative">
   299→          <div className={imvMenu ? "[&>button]:bg-[#ddd0f2]" : ""}>
   300→            <ToolBtn icon={<IconImviron />} label="IMVironments" caret onClick={() => setImvMenu((v) => !v)} active={imv !== "none"} />
   301→          </div>

   303→            <div className="animate-pop absolute right-0 top-full mt-[2px] w-[168px] bg-[#f6f5ec] border border-[#7a57c6] shadow-[3px_3px_8px_rgba(20,5,50,0.4)] py-0.5 z-50">
   304→              {IMV_LIST.map((o) => (

   309→                    setImv(o.id);

   318→            </div>

   322→        <ToolBtn icon={<IconPhotos />} label="Photos" />
   323→      </div>
   324→
   325→      {/* ---- Status strip ---- */}
   326→      <div className="bg-[#f1f0e3] px-3 py-1 flex items-center gap-1.5 text-[11px] text-[#777] border-b border-[#e0ddcd]">
   327→        <StatusDot status={p.contact.status} />

   330→          {p.contact.customMsg ? ` — ${p.contact.customMsg}` : ""}
   331→        </span>
   332→      </div>
   333→
   334→      {/* ---- Messages + sidebar ---- */}
   335→      <div className="flex-1 min-h-0 flex bg-[#f1f0e3] p-1.5 gap-1.5">
   336→        <div
   337→          className="relative flex-1 min-h-0 border border-[#a9a595]"

   339→            background: IMV_BG[imv],
   340→            boxShadow: "inset 1px 1px 2px rgba(0,0,0,0.08)",

   343→          {/* IMVironment ambience — lives BEHIND the message text */}

   345→

   351→          >
   352→            {messages.map((m, i) =>
   353→              m.buzz ? (
   354→                <div key={i} className="animate-msg text-[#d01818] font-bold">
   355→                  BUZZ!!!
   356→                </div>
   357→              ) : m.audible ? (
   358→                <div key={i} className="animate-msg my-1.5 flex items-start gap-2">
   359→                  <span className="font-bold pt-1" style={{ color: m.from === "me" ? "#7b2fbd" : "#2244cc" }}>
   360→                    {m.from === "me" ? ME.handle : p.contact.handle}:
   361→                  </span>
   362→                  <MessageVideo src={m.audible.src} />
   363→                  <span className="pt-1 text-[#333] italic">&quot;{m.audible.caption}&quot;</span>
   364→                </div>
   365→              ) : (
   366→                <div key={i} className="animate-msg break-words">
   367→                  <span className="font-bold" style={{ color: m.from === "me" ? "#7b2fbd" : "#2244cc" }}>
   368→                    {m.from === "me" ? ME.handle : p.contact.handle}:
   369→                  </span>{" "}
   370→                  {m.html ? (
   371→                    <span dangerouslySetInnerHTML={{ __html: m.html }} />
   372→                  ) : (
   373→                    <span>{emojify(m.text ?? "")}</span>
   374→                  )}
   375→                </div>

   377→            )}
   378→          </div>
   379→        </div>
   380→
   381→        {sidebar && (

   383→            {/* friend's picture (top) — fills its panel exactly */}
   384→            <div className="h-[96px] shrink-0 bg-[#cfc6de]">
   385→              {p.contact.avatar ? (
   386→                <img src={p.contact.avatar} alt={p.contact.name} className="w-full h-full object-cover" />
   387→              ) : (
   388→                <div className="w-full h-full bg-[#c3b8d8] flex items-center justify-center">
   389→                  <WhiteFace size={70} />
   390→                </div>
   391→              )}
   392→            </div>
   393→            {/* my picture (bottom of the sidebar, full width) */}
   394→            <div className="flex-1 bg-[#e2dfd2] flex items-end">
   395→              {ME.avatar ? (
   396→                <img src={ME.avatar} alt={ME.name} className="w-full h-[104px] object-cover" />
   397→              ) : (

   402→            </div>

   414→          <button

   417→            title="Show sidebar"

   419→            <IconChevron dir="down" />

   426→        <div className="bg-[#f1f0e3] border-t border-[#d8d5c5] px-2 pt-1 pb-2">

   428→            <span className="text-[11.5px] text-[#555] truncate">{selAudible ?? "Select an Audible below"}</span>

   431→                className="xp-btn h-[20px] px-2.5 text-[11px] text-[#333] flex items-center gap-1"
   432→                onClick={() => setCatMenuOpen((v) => !v)}

   439→                  {AUDIBLE_CATEGORIES.map((c) => (
   440→                    <button
   441→                      key={c.id}

   445→                        setSelAudible(null);

   698→function MessageVideo({ src }: { src: string }) {
   699→  const ref = useRef<HTMLVideoElement>(null);
   700→  const [ended, setEnded] = useState(false);
   701→
   702→  /* plays once, with voice, right after being sent (user-gesture window) */
   703→  useEffect(() => {
   704→    const v = ref.current;
   705→    if (!v) return;
   706→    v.muted = false;
   707→    const t = window.setTimeout(() => v.play().catch(() => {}), 60);
   708→    return () => window.clearTimeout(t);
   709→  }, []);
   710→
   711→  return (
   712→    <button
   713→      className="relative block cursor-pointer group"
   714→      title={ended ? "Click to play again" : "Audible"}
   715→      onClick={() => {
   716→        const v = ref.current;
   717→        if (!v) return;
   718→        if (v.paused || ended) {
   719→          v.currentTime = 0;
   720→          setEnded(false);
   721→          void v.play().catch(() => {});
   722→        } else {
   723→          v.pause();
   724→        }
   725→      }}
   726→    >
   727→      <video
   728→        ref={ref}
   729→        src={src}
   730→        playsInline
   731→        onEnded={() => setEnded(true)}
   732→        className="block w-[120px] h-[90px] object-cover"
   733→      />
   734→      {ended && (
   735→        <span className="absolute inset-0 flex items-center justify-center bg-black/25 group-hover:bg-black/40 transition-colors">
   736→          <svg width="26" height="26" viewBox="0 0 26 26">
   737→            <circle cx="13" cy="13" r="12" fill="rgba(255,255,255,0.9)" />
   738→            <path d="M10 8 L19 13 L10 18 Z" fill="#5a3ca6" />
   739→          </svg>
   740→        </span>
   741→      )}
   742→    </button>
   743→  );
   744→}
