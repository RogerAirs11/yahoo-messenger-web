   169→      /* noop */

   754→function MessageVideo({ src }: { src: string }) {
   755→  const ref = useRef<HTMLVideoElement>(null);
   756→  const [ended, setEnded] = useState(false);
   757→
   758→  /* plays once, with voice, right after being sent (user-gesture window) */
   759→  useEffect(() => {
   760→    const v = ref.current;
   761→    if (!v) return;
   762→    v.muted = false;
   763→    const t = window.setTimeout(() => v.play().catch(() => {}), 60);
   764→    return () => window.clearTimeout(t);
   765→  }, []);
   766→
   767→  return (
   768→    <button
   769→      className="relative block cursor-pointer group"
   770→      title={ended ? "Click to play again" : "Audible"}
   771→      onClick={() => {
   772→        const v = ref.current;
   773→        if (!v) return;
   774→        if (v.paused || ended) {
   775→          v.currentTime = 0;
   776→          setEnded(false);
   777→          void v.play().catch(() => {});
   778→        } else {
   779→          v.pause();
   780→        }
   781→      }}
   782→    >
   783→      <video
   784→        ref={ref}
   785→        src={src}
   786→        playsInline
   787→        onEnded={() => setEnded(true)}
   788→        className="block w-[120px] h-[90px] object-cover"
   789→      />
   790→      {ended && (
   791→        <span className="absolute inset-0 flex items-center justify-center bg-black/25 group-hover:bg-black/40 transition-colors">
   792→          <svg width="26" height="26" viewBox="0 0 26 26">
   793→            <circle cx="13" cy="13" r="12" fill="rgba(255,255,255,0.9)" />
   794→            <path d="M10 8 L19 13 L10 18 Z" fill="#5a3ca6" />
   795→          </svg>
   796→        </span>
   797→      )}
   798→    </button>
   799→  );
   800→}
