   358→export function emojify(text: string): React.ReactNode[] {
   359→  const nodes: React.ReactNode[] = [];
   360→  let rest = text;
   361→  let key = 0;
   362→  while (rest.length) {
   363→    let earliest = -1;
   364→    let matched: Emo | null = null;
   365→    for (const e of EMO) {
   366→      const i = rest.indexOf(e.code);
   367→      if (i !== -1 && (earliest === -1 || i < earliest || (i === earliest && matched && e.code.length > matched.code.length))) {
   368→        earliest = i;
   369→        matched = e;
   370→      }
   371→    }
   372→    if (!matched || earliest === -1) {
   373→      nodes.push(rest);
   374→      break;
   375→    }
   376→    if (earliest > 0) nodes.push(rest.slice(0, earliest));
   377→    nodes.push(<Emoticon key={key++} file={matched.file} />);
   378→    rest = rest.slice(earliest + matched.code.length);
   379→  }
   380→  return nodes;
   381→}

   383→/* Replace emoticon codes inside a rich-HTML string without touching tags.
   384→   EMO is sorted longest-first, so :(( is replaced before :( can bite it. */
   385→export function emojifyHtml(html: string): string {
   386→  return html
   387→    .split(/(<[^>]*>)/g)
   388→    .map((seg) => {
