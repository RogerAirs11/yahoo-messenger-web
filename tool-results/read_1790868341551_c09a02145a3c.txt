   324→export function emojify(text: string): React.ReactNode[] {
   325→  const nodes: React.ReactNode[] = [];
   326→  let rest = text;
   327→  let key = 0;
   328→  while (rest.length) {
   329→    let earliest = -1;
   330→    let matched: Emo | null = null;
   331→    for (const e of EMO) {
   332→      const i = rest.indexOf(e.code);
   333→      if (i !== -1 && (earliest === -1 || i < earliest || (i === earliest && matched && e.code.length > matched.code.length))) {
   334→        earliest = i;
   335→        matched = e;
   336→      }
   337→    }
   338→    if (!matched || earliest === -1) {
   339→      nodes.push(rest);
   340→      break;
   341→    }
   342→    if (earliest > 0) nodes.push(rest.slice(0, earliest));
   343→    nodes.push(<Emoticon key={key++} file={matched.file} />);
   344→    rest = rest.slice(earliest + matched.code.length);
   345→  }
   346→  return nodes;
   347→}

   349→/* Replace emoticon codes inside a rich-HTML string without touching tags. */
   350→export function emojifyHtml(html: string): string {
   351→  return html
   352→    .split(/(<[^>]*>)/g)
   353→    .map((seg) => {
   354→      if (seg.startsWith("<")) return seg;
   355→      let out = seg.replace(/&lt;3/g, `<img src="${EMO_GIF("love")}" class="inline-block align-[-4px]">`);
   356→      for (const e of EMO) {
   357→        if (e.code.includes("<") || e.code.includes(">")) continue;
   358→        if (!out.includes(e.code)) continue;
   359→        out = out.split(e.code).join(`<img src="${EMO_GIF(e.file)}" class="inline-block align-[-4px]" alt="${e.name}">`);
   360→      }
   361→      return out;
   362→    })
   363→    .join("");
   364→}
