    31→/* deterministic PRNG so layouts are stable across re-renders / SSR */
    32→function seeded(seed: number) {
    33→  let s = seed >>> 0;
    34→  return () => {
    35→    s = (s * 1664525 + 1013904223) >>> 0;
    36→    return s / 4294967296;
    37→  };
    38→}
