   698→   698→function MessageVideo({ src }: { src: string }) {
   699→   699→  const ref = useRef<HTMLVideoElement>(null);
   700→   700→  const [ended, setEnded] = useState(false);
   701→   701→
   702→   702→  /* plays once, with voice, right after being sent (user-gesture window) */
   703→   703→  useEffect(() => {
   704→   704→    const v = ref.current;
   705→   705→    if (!v) return;
   706→   706→    v.muted = false;
   707→   707→    const t = window.setTimeout(() => v.play().catch(() => {}), 60);
   708→   708→    return () => window.clearTimeout(t);
   709→   709→  }, []);
   710→   710→
   711→   711→  return (
   712→   712→    <button
   713→   713→      className="relative block cursor-pointer group"
   714→   714→      title={ended ? "Click to play again" : "Audible"}
   715→   715→      onClick={() => {
   716→   716→        const v = ref.current;
   717→   717→        if (!v) return;
   718→   718→        if (v.paused || ended) {
   719→   719→          v.currentTime = 0;
   720→   720→          setEnded(false);
   721→   721→          void v.play().catch(() => {});
   722→   722→        } else {
   723→   723→          v.pause();
   724→   724→        }
   725→   725→      }}
   726→   726→    >
   727→   727→      <video
   728→   728→        ref={ref}
   729→   729→        src={src}
   730→   730→        playsInline
   731→   731→        onEnded={() => setEnded(true)}
   732→   732→        className="block w-[120px] h-[90px] object-cover"
   733→   733→      />
   734→   734→      {ended && (
   735→   735→        <span className="absolute inset-0 flex items-center justify-center bg-black/25 group-hover:bg-black/40 transition-colors">
   736→   736→          <svg width="26" height="26" viewBox="0 0 26 26">
   737→   737→            <circle cx="13" cy="13" r="12" fill="rgba(255,255,255,0.9)" />
   738→   738→            <path d="M10 8 L19 13 L10 18 Z" fill="#5a3ca6" />
   739→   739→          </svg>
   740→   740→        </span>
   741→   741→      )}
   742→   742→    </button>
   743→   743→  );
   744→   744→}
